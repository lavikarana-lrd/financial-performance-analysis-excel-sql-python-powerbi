## Hypothesis testing ##
## 1. ANOVA: Does mean monthly expense differ significantly across business units
expense_df= df_FT[df_FT['Account_type']=='Expense'].copy()
expense_df['Transaction_date']= pd.to_datetime(expense_df['Transaction_date'])
expense_df['YearMonth'] = expense_df['Transaction_date'].dt.to_period('M')
expense_df['Amount']= pd.to_numeric(expense_df['Amount'], errors='coerce')
monthly_exp_by_unit = expense_df.groupby(['YearMonth','Business_unit',])['Amount'].sum().reset_index()

units = monthly_exp_by_unit['Business_unit'].unique()
grouped_data= [monthly_exp_by_unit[monthly_exp_by_unit['Business_unit'] == u]['Amount'] for u in units]
f_stat,p_value= stats.f_oneway(*grouped_data)
f_stat= f_stat.round(2)
p_value= p_value.round(2)

print("F_stat:", f_stat)
print("p_value:", p_value)
los= 0.05
if p_value < los:
    print("\nHo is rejected. Mean monthly expense differ significantly across business units.")
else:
    print("\nHo is accepted. Mean monthly expense does not differ significantly across business units.")

#2. T-Test: Compare average transaction revenue between East and West regions
revenue_df= df_FT[df_FT['Account_type']=='Revenue'].copy()

south_rev= revenue_df[revenue_df['Region']=='South']['Amount']
t_stat, p_value = stats.ttest_ind(north_rev, south_rev, equal_var=False)

# 4. Chi-Square Test: Is there an association between Customer Segment and Activity Status?
from scipy.stats import chi2_contingency

# Step 1: Create contingency table
contingency_table = pd.crosstab(df_Customers['segment'], df_Customers['Status'])

print("Contingency Table:\n", contingency_table)

# Step 2: Perform Chi-Square Test
chi2, p_value, dof, expected = chi2_contingency(contingency_table)

# Step 3: Output results
print("Chi-square statistic:", round(chi2, 2))
print("p-value:", round(p_value, 4))
print("Degrees of freedom:", dof)
if p_value < 0.05:
    print("Ho is rejected: There is an association between Customer Segment and Activity Status.")
else:
    print("Ho is accepted: There is no association between Customer Segment and Activity Status.")
